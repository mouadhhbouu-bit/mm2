local function CreateSlider(parent, name, minVal, maxVal, defaultVal, callback)
	local frame = Instance.new("Frame")
	frame.Size = UDim2.new(0.96, 0, 0, 38)
	frame.BackgroundColor3 = Color3.fromRGB(12, 20, 35)
	frame.BackgroundTransparency = 0.3
	frame.Parent = parent
	Instance.new("UICorner", frame).CornerRadius = UDim.new(0, 6)

	local titleLbl = Instance.new("TextLabel")
	titleLbl.Size = UDim2.new(0.6, 0, 0, 16)
	titleLbl.Position = UDim2.new(0, 8, 0, 2)
	titleLbl.BackgroundTransparency = 1
	titleLbl.Text = name
	titleLbl.TextColor3 = Color3.fromRGB(220, 230, 245)
	titleLbl.Font = Enum.Font.FredokaOne
	titleLbl.TextSize = 10
	titleLbl.TextXAlignment = Enum.TextXAlignment.Left
	titleLbl.Parent = frame

	local valLbl = Instance.new("TextLabel")
	valLbl.Size = UDim2.new(0.3, 0, 0, 16)
	valLbl.Position = UDim2.new(0.68, 0, 0, 2)
	valLbl.BackgroundTransparency = 1
	valLbl.Text = tostring(defaultVal)
	valLbl.TextColor3 = Color3.fromRGB(0, 153, 255)
	valLbl.Font = Enum.Font.FredokaOne
	valLbl.TextSize = 10
	valLbl.TextXAlignment = Enum.TextXAlignment.Right
	valLbl.Parent = frame

	local sliderBar = Instance.new("Frame")
	sliderBar.Size = UDim2.new(0.92, 0, 0, 4)
	sliderBar.Position = UDim2.new(0.04, 0, 0.7, 0)
	sliderBar.BackgroundColor3 = Color3.fromRGB(25, 35, 55)
	sliderBar.Parent = frame
	Instance.new("UICorner", sliderBar).CornerRadius = UDim.new(1, 0)

	local fill = Instance.new("Frame")
	fill.Size = UDim2.new((defaultVal - minVal)/(maxVal - minVal), 0, 1, 0)
	fill.BackgroundColor3 = Color3.fromRGB(0, 153, 255)
	fill.Parent = sliderBar
	Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

	local isDragging = false
	local function update(input)
		local pos = math.clamp((input.Position.X - sliderBar.AbsolutePosition.X) / sliderBar.AbsoluteSize.X, 0, 1)
		fill.Size = UDim2.new(pos, 0, 1, 0)
		local value = math.floor(minVal + (maxVal - minVal) * pos)
		valLbl.Text = tostring(value)
		callback(value)
	end

	sliderBar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			isDragging = true update(input)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			isDragging = false
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if isDragging and (input.UserInputType == Enum.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			update(input)
		end
	end)
end

local espSettings = {Murderer = false, Sheriff = false, Players = false, DroppedWeapon = false}

local function applyHighlight(target, color, name)
	if target and target:FindFirstChild("HumanoidRootPart") then
		local hl = target:FindFirstChild(name) or Instance.new("Highlight")
		hl.Name = name
		hl.FillColor = color
		hl.OutlineColor = Color3.fromRGB(255, 255, 255)
		hl.FillTransparency = 0.4
		hl.Parent = target
	end
end

task.spawn(function()
	while task.wait(0.5) do
		for _, plr in pairs(Players:GetPlayers()) do
			if plr ~= LocalPlayer and plr.Character then
				local char = plr.Character
				local bp = plr:FindFirstChild("Backpack")
				
				local hasKnife = (bp and bp:FindFirstChild("Knife")) or char:FindFirstChild("Knife")
				local hasGun = (bp and (bp:FindFirstChild("Gun") or bp:FindFirstChild("Revolver"))) or char:FindFirstChild("Gun") or char:FindFirstChild("Revolver")

				if hasKnife and espSettings.Murderer then
					applyHighlight(char, Color3.fromRGB(255, 30, 30), "MouadhESP")
				elseif hasGun and espSettings.Sheriff then
					applyHighlight(char, Color3.fromRGB(30, 120, 255), "MouadhESP")
				elseif not hasKnife and not hasGun and espSettings.Players then
					applyHighlight(char, Color3.fromRGB(30, 255, 100), "MouadhESP")
				else
					if char:FindFirstChild("MouadhESP") then char.MouadhESP:Destroy() end
				end
			end
		end

		if espSettings.DroppedWeapon then
			for _, item in pairs(Workspace:GetDescendants()) do
				if item:IsA("Tool") and (item.Name == "GunDrop" or item.Name == "Gun") and not item:IsDescendantOf(Players) then
					local hl = item:FindFirstChild("MouadhWeaponESP") or Instance.new("Highlight")
					hl.Name = "MouadhWeaponESP"
					hl.FillColor = Color3.fromRGB(255, 225, 0)
					hl.FillTransparency = 0.2
					hl.Parent = item
				end
			end
		end
	end
end)

local autoFarmActive = false

local function getCoins()
	local coinList = {}
	for _, obj in pairs(Workspace:GetDescendants()) do
		if (obj.Name == "Coin" or obj.Name == "CoinContainer" or obj.Name == "CoinServer") and obj:IsA("BasePart") then
			if obj.Transparency < 0.9 then table.insert(coinList, obj) end
		elseif obj:IsA("Model") and (obj.Name == "Coin" or obj.Name == "GoldCoin") then
			local mainPart = obj:FindFirstChildWhichIsA("BasePart")
			if mainPart and mainPart.Transparency < 0.9 then table.insert(coinList, mainPart) end
		end
	end
	return coinList
end

task.spawn(function()
	while task.wait(0.2) do
		if autoFarmActive and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
			local hrp = LocalPlayer.Character.HumanoidRootPart
			local coins = getCoins()
			
			for _, part in pairs(LocalPlayer.Character:GetDescendants()) do
				if part:IsA("BasePart") then part.CanCollide = false end
			end

			for _, coinPart in pairs(coins) do
				if autoFarmActive and coinPart and coinPart.Parent and coinPart.Transparency < 0.9 then
					local underPos = coinPart.CFrame * CFrame.new(0, -7.5, 0)
					local dist = (hrp.Position - underPos.Position).Magnitude
					local tweenTime = math.clamp(dist / 45, 0.15, 0.8)
					
					local tween = TweenService:Create(hrp, TweenInfo.new(tweenTime, Enum.EasingStyle.Linear), {CFrame = underPos})
					tween:Play()
					tween.Completed:Wait()
					
					hrp.CFrame = coinPart.CFrame
					task.wait(0.12)
				end
			end
		end
	end
end)

local flingActive = false
local flingAllActive = false

local function primaryFling(targetChar)
	local char = LocalPlayer.Character
	if not char or not char:FindFirstChild("HumanoidRootPart") or not targetChar or not targetChar:FindFirstChild("HumanoidRootPart") then return end

	local hrp = char.HumanoidRootPart
	local originalPos = hrp.CFrame
	local targetHrp = targetChar.HumanoidRootPart
	
	local bv = Instance.new("BodyVelocity")
	bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
	bv.Velocity = Vector3.new(999999, 999999, 999999)
	bv.Parent = hrp

	local bav = Instance.new("BodyAngularVelocity")
	bav.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
	bav.AngularVelocity = Vector3.new(999999, 999999, 999999)
	bav.Parent = hrp

	for _, p in pairs(char:GetDescendants()) do
		if p:IsA("BasePart") then p.CanCollide = false end
	end

	local startTime = tick()
	while (tick() - startTime) < 2 and targetChar and targetHrp and (flingActive or flingAllActive) do
		hrp.CFrame = targetHrp.CFrame * CFrame.new(math.random(-1, 1), 0, math.random(-1, 1))
		task.wait()
	end

	bv:Destroy()
	bav:Destroy()
	
	if hrp and originalPos then
		hrp.CFrame = originalPos
	end
end
